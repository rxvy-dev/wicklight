cmake_minimum_required(VERSION 3.10)
project(wicklight CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
file(GLOB SRC src/*.cpp)
add_executable(wicklight ${SRC})
target_include_directories(wicklight PRIVATE src third_party/include)
target_link_libraries(wicklight ${CMAKE_DL_LIBS})
install(TARGETS wicklight DESTINATION bin)
install(FILES packaging/wicklight.desktop DESTINATION share/applications)
install(FILES packaging/wicklight.png DESTINATION share/icons/hicolor/256x256/apps)
